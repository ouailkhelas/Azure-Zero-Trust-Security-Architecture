# Azure Zero Trust Security - RBAC Commands
# Role assignment and permissions management

## Configuration Variables
RESOURCE_GROUP="rg-zerotrust-security"
STORAGE_ACCOUNT="stzerotrustdev"
KEY_VAULT="kv-zerotrust-dev"
USER_EMAIL="user@example.com"  # Replace with actual email

# ==========================================
# GET USER/GROUP OBJECT ID
# ==========================================

## Get current logged-in user object ID
az ad signed-in-user show --query id -o tsv

## Get specific user object ID by email
USER_ID=$(az ad user show --id "$USER_EMAIL" --query id -o tsv)

## Get group object ID by name
GROUP_ID=$(az ad group show --group "GroupName" --query id -o tsv)

# ==========================================
# STORAGE ACCOUNT ROLES
# ==========================================

## Assign Storage Blob Data Owner (full control)
az role assignment create \
  --role "Storage Blob Data Owner" \
  --assignee "$USER_ID" \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.Storage/storageAccounts/$STORAGE_ACCOUNT"

## Assign Storage Blob Data Contributor (read/write)
az role assignment create \
  --role "Storage Blob Data Contributor" \
  --assignee "$USER_ID" \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.Storage/storageAccounts/$STORAGE_ACCOUNT"

## Assign Storage Blob Data Reader (read-only)
az role assignment create \
  --role "Storage Blob Data Reader" \
  --assignee "$USER_ID" \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.Storage/storageAccounts/$STORAGE_ACCOUNT"

## View assigned roles on storage account
az role assignment list \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.Storage/storageAccounts/$STORAGE_ACCOUNT"

# ==========================================
# KEY VAULT ROLES
# ==========================================

## Assign Key Vault Administrator (full control)
az role assignment create \
  --role "Key Vault Administrator" \
  --assignee "$USER_ID" \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.KeyVault/vaults/$KEY_VAULT"

## Assign Key Vault Secrets Officer (manage secrets)
az role assignment create \
  --role "Key Vault Secrets Officer" \
  --assignee "$USER_ID" \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.KeyVault/vaults/$KEY_VAULT"

## Assign Key Vault Secrets User (read secrets only)
az role assignment create \
  --role "Key Vault Secrets User" \
  --assignee "$USER_ID" \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.KeyVault/vaults/$KEY_VAULT"

## View assigned roles on Key Vault
az role assignment list \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.KeyVault/vaults/$KEY_VAULT"

# ==========================================
# RESOURCE GROUP LEVEL ROLES
# ==========================================

## Assign Contributor role (read/write to all resources)
az role assignment create \
  --role "Contributor" \
  --assignee "$USER_ID" \
  --resource-group "$RESOURCE_GROUP"

## Assign Reader role (read-only)
az role assignment create \
  --role "Reader" \
  --assignee "$USER_ID" \
  --resource-group "$RESOURCE_GROUP"

## Assign Owner role (full control - use with caution)
az role assignment create \
  --role "Owner" \
  --assignee "$USER_ID" \
  --resource-group "$RESOURCE_GROUP"

# ==========================================
# VIEW AND MANAGE ROLE ASSIGNMENTS
# ==========================================

## List all role assignments in resource group
az role assignment list \
  --resource-group "$RESOURCE_GROUP" \
  --output table

## List role assignments for specific user
az role assignment list \
  --assignee "$USER_ID" \
  --output table

## List role assignments for specific scope
az role assignment list \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP"

## Get details of specific role assignment
ASSIGNMENT_ID="xxxxx"  # Replace with actual assignment ID
az role assignment show \
  --ids "$ASSIGNMENT_ID"

# ==========================================
# REMOVE ROLE ASSIGNMENTS
# ==========================================

## Remove role assignment
az role assignment delete \
  --role "Contributor" \
  --assignee "$USER_ID" \
  --resource-group "$RESOURCE_GROUP"

## Remove all role assignments for user in resource group
az role assignment list \
  --assignee "$USER_ID" \
  --resource-group "$RESOURCE_GROUP" \
  --query "[].id" -o tsv | xargs -I {} az role assignment delete --ids {}

# ==========================================
# CUSTOM RBAC ROLES
# ==========================================

## Create custom role (advanced)
# First create JSON file with role definition
cat > custom-role.json << 'EOF'
{
  "Name": "Custom Secure Reader",
  "Description": "Can read resources but cannot modify",
  "Actions": [
    "*/read"
  ],
  "NotActions": [],
  "AssignableScopes": [
    "/subscriptions/{subscription-id}"
  ]
}
EOF

az role definition create --role-definition @custom-role.json

# ==========================================
# POLICY ASSIGNMENTS
# ==========================================

## Assign built-in policy (require HTTPS)
az policy assignment create \
  --name "require-https-policy" \
  --display-name "Require HTTPS" \
  --policy "e9c8013f-fd6d-48ba-9a32-6dd773673e6e" \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP"

## List policy assignments
az policy assignment list \
  --resource-group "$RESOURCE_GROUP"

# ==========================================
# CHECK PERMISSIONS
# ==========================================

## Check if user has specific permission
az role assignment list \
  --assignee "$USER_ID" \
  --scope "/subscriptions/$(az account show --query id -o tsv)" \
  --include-inherited \
  --query "[].roleDefinitionName" -o tsv

## Test access to storage account
az storage account show \
  --name "$STORAGE_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP"

## Test access to Key Vault
az keyvault show \
  --name "$KEY_VAULT" \
  --resource-group "$RESOURCE_GROUP"

# ==========================================
# AUDIT LOGGING
# ==========================================

## View activity log for role assignments
az monitor activity-log list \
  --resource-group "$RESOURCE_GROUP" \
  --query "[?operationName.value == 'Microsoft.Authorization/roleAssignments/write']" \
  --output table

## View all activity in resource group
az monitor activity-log list \
  --resource-group "$RESOURCE_GROUP" \
  --output table

# ==========================================
# SECURITY RECOMMENDATIONS
# ==========================================

## Enable managed identity (for resources)
az identity create \
  --name "zerotrust-identity" \
  --resource-group "$RESOURCE_GROUP"

## Assign role to managed identity
IDENTITY_ID=$(az identity show \
  --name "zerotrust-identity" \
  --resource-group "$RESOURCE_GROUP" \
  --query "principalId" -o tsv)

az role assignment create \
  --role "Storage Blob Data Reader" \
  --assignee "$IDENTITY_ID" \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.Storage/storageAccounts/$STORAGE_ACCOUNT"
