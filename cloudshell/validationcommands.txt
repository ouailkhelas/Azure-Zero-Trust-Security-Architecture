# Azure Zero Trust Security - Validation Commands
# Test secure access and verify deployment

## Configuration
RESOURCE_GROUP="rg-zerotrust-security"
STORAGE_ACCOUNT="stzerotrustdev"
KEY_VAULT="kv-zerotrust-dev"
VNET_NAME="vnet-zerotrust-dev"

# ==========================================
# VERIFY DEPLOYMENT
# ==========================================

## List all resources in resource group
az resource list \
  --resource-group "$RESOURCE_GROUP" \
  --output table

## Check resource group exists
az group show --name "$RESOURCE_GROUP"

## List deployments in resource group
az deployment group list \
  --resource-group "$RESOURCE_GROUP" \
  --output table

# ==========================================
# STORAGE ACCOUNT VALIDATION
# ==========================================

## Check storage account properties
az storage account show \
  --name "$STORAGE_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP" \
  --query "{}.[id, name, kind, supportsHttpsTrafficOnly, networkAcls]"

## Verify HTTPS only enabled
az storage account show \
  --name "$STORAGE_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP" \
  --query "supportsHttpsTrafficOnly"

## Verify public blob access disabled
az storage account show \
  --name "$STORAGE_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP" \
  --query "allowBlobPublicAccess"

## Verify network ACLs (should be Deny by default)
az storage account show \
  --name "$STORAGE_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP" \
  --query "networkAcls.defaultAction"

## List containers in storage account
az storage container list \
  --account-name "$STORAGE_ACCOUNT" \
  --output table

## Check TLS version
az storage account show \
  --name "$STORAGE_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP" \
  --query "minimumTlsVersion"

## Test blob access (should be restricted)
az storage blob list \
  --account-name "$STORAGE_ACCOUNT" \
  --container-name "secure-data" \
  --output table

# ==========================================
# KEY VAULT VALIDATION
# ==========================================

## Check Key Vault properties
az keyvault show \
  --name "$KEY_VAULT" \
  --resource-group "$RESOURCE_GROUP"

## Verify RBAC enabled
az keyvault show \
  --name "$KEY_VAULT" \
  --resource-group "$RESOURCE_GROUP" \
  --query "properties.enableRbacAuthorization"

## Verify network ACLs
az keyvault show \
  --name "$KEY_VAULT" \
  --resource-group "$RESOURCE_GROUP" \
  --query "properties.networkAcls"

## Verify soft delete enabled
az keyvault show \
  --name "$KEY_VAULT" \
  --resource-group "$RESOURCE_GROUP" \
  --query "properties.enableSoftDelete"

## Verify purge protection enabled
az keyvault show \
  --name "$KEY_VAULT" \
  --resource-group "$RESOURCE_GROUP" \
  --query "properties.purgeProtectionEnabled"

## List secrets in Key Vault
az keyvault secret list \
  --vault-name "$KEY_VAULT" \
  --output table

## Get specific secret (test access)
az keyvault secret show \
  --vault-name "$KEY_VAULT" \
  --name "demo-secret" \
  --query "value"

# ==========================================
# VIRTUAL NETWORK VALIDATION
# ==========================================

## Check Virtual Network properties
az network vnet show \
  --name "$VNET_NAME" \
  --resource-group "$RESOURCE_GROUP"

## List subnets
az network vnet subnet list \
  --vnet-name "$VNET_NAME" \
  --resource-group "$RESOURCE_GROUP" \
  --output table

## Check subnet details
az network vnet subnet show \
  --vnet-name "$VNET_NAME" \
  --name "subnet-secure-dev" \
  --resource-group "$RESOURCE_GROUP"

## List Network Security Groups
az network nsg list \
  --resource-group "$RESOURCE_GROUP" \
  --output table

## Check NSG rules
az network nsg rule list \
  --nsg-name "nsg-vnet-zerotrust-dev" \
  --resource-group "$RESOURCE_GROUP" \
  --output table

## Verify service endpoints (Storage, KeyVault)
az network vnet subnet show \
  --vnet-name "$VNET_NAME" \
  --name "subnet-secure-dev" \
  --resource-group "$RESOURCE_GROUP" \
  --query "serviceEndpoints"

# ==========================================
# SECURITY VALIDATION
# ==========================================

## Check if resource has Managed Identity
az resource show \
  --name "$STORAGE_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP" \
  --resource-type "Microsoft.Storage/storageAccounts" \
  --query "identity"

## List all role assignments in resource group
az role assignment list \
  --resource-group "$RESOURCE_GROUP" \
  --output table

## Check encryption status
az storage account encryption-scope list \
  --account-name "$STORAGE_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP"

## Verify minimum TLS version is 1.2
az storage account show \
  --name "$STORAGE_ACCOUNT" \
  --resource-group "$RESOURCE_GROUP" \
  --query "minimumTlsVersion" | grep -i "TLS1_2" && echo "✅ TLS 1.2 enforced" || echo "❌ TLS version not enforced"

# ==========================================
# ACCESS CONTROL TESTING
# ==========================================

## Test current user permissions on storage
az role assignment list \
  --assignee "$(az ad signed-in-user show --query id -o tsv)" \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.Storage/storageAccounts/$STORAGE_ACCOUNT"

## Test current user permissions on Key Vault
az role assignment list \
  --assignee "$(az ad signed-in-user show --query id -o tsv)" \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.KeyVault/vaults/$KEY_VAULT"

## List all users with access to Key Vault
az role assignment list \
  --scope "/subscriptions/$(az account show --query id -o tsv)/resourceGroups/$RESOURCE_GROUP/providers/Microsoft.KeyVault/vaults/$KEY_VAULT" \
  --output table

# ==========================================
# AUDIT AND COMPLIANCE
# ==========================================

## View activity log (past 1 day)
az monitor activity-log list \
  --resource-group "$RESOURCE_GROUP" \
  --start-time "$(date -u -d '-1 day' '+%Y-%m-%dT%H:%M:%SZ')" \
  --output table

## Check for failed sign-ins (Entra ID)
# Note: Requires appropriate permissions
az ad sign-in-logs list --output table | head -20

## List storage account access logs
az storage logging show \
  --account-name "$STORAGE_ACCOUNT"

# ==========================================
# DIAGNOSTIC CHECKS
# ==========================================

## Check Key Vault network connectivity
az keyvault network-rule list \
  --vault-name "$KEY_VAULT" \
  --resource-group "$RESOURCE_GROUP"

## Test storage account network access
curl -I "https://$STORAGE_ACCOUNT.blob.core.windows.net"

## Check Key Vault URI is accessible
curl -I "https://$KEY_VAULT.vault.azure.net"

## Verify NSG allows HTTPS (443)
az network nsg rule list \
  --nsg-name "nsg-vnet-zerotrust-dev" \
  --resource-group "$RESOURCE_GROUP" \
  --query "[?contains(name, 'HTTPS')]" \
  --output table

# ==========================================
# COMPLIANCE VERIFICATION
# ==========================================

## Check all resources have tags
az resource list \
  --resource-group "$RESOURCE_GROUP" \
  --query "[].{Name:name, Tags:tags}" \
  --output table

## Verify resources follow naming convention
az resource list \
  --resource-group "$RESOURCE_GROUP" \
  --output table | grep -E "^(st|kv|vnet|nsg)" && echo "✅ Naming conventions followed"

## Generate compliance report
echo "=== Zero Trust Security Compliance Report ===" > compliance-report.txt
echo "Deployment Time: $(date)" >> compliance-report.txt
echo "" >> compliance-report.txt
echo "Resource Count:" >> compliance-report.txt
az resource list --resource-group "$RESOURCE_GROUP" --query "length(@)" >> compliance-report.txt
echo "" >> compliance-report.txt
echo "HTTPS Enforcement:" >> compliance-report.txt
az storage account show --name "$STORAGE_ACCOUNT" --resource-group "$RESOURCE_GROUP" --query "supportsHttpsTrafficOnly" >> compliance-report.txt
echo "" >> compliance-report.txt
echo "Network Isolation:" >> compliance-report.txt
az storage account show --name "$STORAGE_ACCOUNT" --resource-group "$RESOURCE_GROUP" --query "networkAcls.defaultAction" >> compliance-report.txt

echo "Report saved to: compliance-report.txt"
cat compliance-report.txt

# ==========================================
# HEALTH CHECK SUMMARY
# ==========================================

echo ""
echo "=== Zero Trust Security Health Check ==="
echo ""

# Check storage account
if az storage account show --name "$STORAGE_ACCOUNT" --resource-group "$RESOURCE_GROUP" &>/dev/null; then
  echo "✅ Storage Account: Healthy"
else
  echo "❌ Storage Account: Not found"
fi

# Check Key Vault
if az keyvault show --name "$KEY_VAULT" --resource-group "$RESOURCE_GROUP" &>/dev/null; then
  echo "✅ Key Vault: Healthy"
else
  echo "❌ Key Vault: Not found"
fi

# Check Virtual Network
if az network vnet show --name "$VNET_NAME" --resource-group "$RESOURCE_GROUP" &>/dev/null; then
  echo "✅ Virtual Network: Healthy"
else
  echo "❌ Virtual Network: Not found"
fi

echo ""
echo "All checks completed!"
